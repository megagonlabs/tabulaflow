Answer: The customer summary and total revenue bar chart are shown above. The joined results are also saved as `customer_sales_support_summary` in the workspace.

------------------------------------------------------------------------
Artifact: customer summary
Source: workspace
SQL: CREATE OR REPLACE TABLE customer_sales_support_summary AS
WITH revenue AS (
    SELECT
        customer_id,
        SUM(revenue_usd) AS total_revenue_usd
    FROM sales_orders
    GROUP BY customer_id
),
open_high_priority AS (
    SELECT
        customer_id,
        COUNT(*) FILTER (
            WHERE lower(status) = 'open'
              AND lower(priority) = 'high'
        ) AS open_high_priority_tickets
    FROM support_tickets
    GROUP BY customer_id
),
customers AS (
    SELECT customer_id FROM revenue
    UNION
    SELECT customer_id FROM open_high_priority
)
SELECT
    c.customer_id,
    COALESCE(r.total_revenue_usd, 0) AS total_revenue_usd,
    COALESCE(t.open_high_priority_tickets, 0) AS open_high_priority_tickets
FROM customers c
LEFT JOIN revenue r USING (customer_id)
LEFT JOIN open_high_priority t USING (customer_id)
ORDER BY c.customer_id;

SELECT * FROM customer_sales_support_summary ORDER BY customer_id;
