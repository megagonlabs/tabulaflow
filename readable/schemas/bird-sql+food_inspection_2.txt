-- Database: food_inspection_2

-- Table: employee (75 rows)
CREATE TABLE employee (
    employee_id INTEGER PRIMARY KEY,  -- e.g. 103705
    first_name TEXT,  -- e.g. 'Anastasia'
    last_name TEXT,  -- e.g. 'Hansen'
    address TEXT,  -- e.g. '6023 S Elizabeth St'
    city TEXT,  -- values: {'Chicago', 'Hoffman Estates', 'Park Forest'}
    state TEXT,  -- values: {'IL'}
    zip INTEGER,  -- e.g. 60636
    phone TEXT,  -- e.g. '(773) 424-8729'
    title TEXT,  -- values: {'Division Manager', 'Sanitarian', 'Supervisor'}
    salary INTEGER,  -- e.g. 79300
    supervisor INTEGER,  -- e.g. 177316; FK -> employee.employee_id
    FOREIGN KEY (supervisor) REFERENCES employee(employee_id)
);

-- Table: establishment (31642 rows)
CREATE TABLE establishment (
    license_no INTEGER PRIMARY KEY,  -- e.g. 1
    dba_name TEXT,  -- e.g. 'HARVEST CRUSADES MINISTRIES'
    aka_name TEXT,  -- e.g. 'COSI'
    facility_type TEXT,  -- e.g. 'Special Event'
    risk_level INTEGER,  -- e.g. 2
    address TEXT,  -- e.g. '118 N CENTRAL AVE '
    city TEXT,  -- values: {'ALSIP  ', 'BERWYN ', 'BURNHAM', 'CHICAGO', 'CHicago', 'CICERO ', 'Chicago', 'GLENCOE', 'JUSTICE', 'LOMBARD', 'MAYWOOD', 'Maywood', 'SKOKIE ', 'SUMMIT ', 'WORTH  ', 'chicago'}
    state TEXT,  -- values: {'IL'}
    zip INTEGER,  -- e.g. 60644
    latitude REAL,  -- e.g. 41.883
    longitude REAL,  -- e.g. -87.765
    ward INTEGER  -- e.g. 29
);

-- Table: inspection (143870 rows)
CREATE TABLE inspection (
    inspection_id INTEGER PRIMARY KEY,  -- e.g. 44247
    inspection_date DATE,  -- e.g. '2010-01-05'
    inspection_type TEXT,  -- e.g. 'Complaint'
    results TEXT,  -- values: {'Business Not Located', 'Fail', 'No Entry', 'Not Ready', 'Out of Business', 'Pass w/ Conditions', 'Pass'}
    employee_id INTEGER,  -- e.g. 141319; FK -> employee.employee_id
    license_no INTEGER,  -- e.g. 1222441; FK -> establishment.license_no
    followup_to INTEGER,  -- e.g. 67871; FK -> inspection.inspection_id
    FOREIGN KEY (employee_id) REFERENCES employee(employee_id),
    FOREIGN KEY (license_no) REFERENCES establishment(license_no),
    FOREIGN KEY (followup_to) REFERENCES inspection(inspection_id)
);

-- Table: inspection_point (46 rows)
CREATE TABLE inspection_point (
    point_id INTEGER PRIMARY KEY,  -- e.g. 1
    Description TEXT,  -- e.g. 'Source sound condition, no spoilage, foods properly labeled, shellfish tags in place'
    category TEXT,  -- e.g. 'Food Protection'
    code TEXT,  -- e.g. '7-38-005 (B) (B-2)'
    fine INTEGER,  -- e.g. 500
    point_level TEXT  -- values: {'Critical', 'Minor   ', 'Serious '}
);

-- Table: violation (525709 rows)
CREATE TABLE violation (
    inspection_id INTEGER,  -- e.g. 44247; FK -> inspection.inspection_id
    point_id INTEGER,  -- e.g. 30; FK -> inspection_point.point_id
    fine INTEGER,  -- e.g. 100
    inspector_comment TEXT,  -- e.g. 'All food not stored in the original container shal...AND DATE.  MUST LABEL AND DATE ALL PREPARED FOODS.'
    PRIMARY KEY (inspection_id, point_id),
    FOREIGN KEY (inspection_id) REFERENCES inspection(inspection_id),
    FOREIGN KEY (point_id) REFERENCES inspection_point(point_id)
);
