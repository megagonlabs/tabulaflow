-- Database: genes

-- Table: Classification (862 rows)
CREATE TABLE Classification (
    GeneID TEXT NOT NULL PRIMARY KEY,  -- e.g. 'G234064'
    Localization TEXT NOT NULL  -- e.g. 'cytoplasm'
);

-- Table: Genes (4346 rows)
CREATE TABLE Genes (
    GeneID TEXT NOT NULL,  -- e.g. 'G234064'; FK -> Classification.GeneID
    Essential TEXT NOT NULL,  -- values: {'?', 'Ambiguous-Essential', 'Essential', 'Non-Essential'}
    Class TEXT NOT NULL,  -- e.g. 'GTP/GDP-exchange factors (GEFs)'
    Complex TEXT,  -- e.g. 'Translation complexes'
    Phenotype TEXT NOT NULL,  -- values: {'?', 'Auxotrophies, carbon and', 'Carbohydrate and lipid biosynthesis defects', 'Cell cycle defects', 'Cell morphology and organelle mutants', 'Conditional phenotypes', 'Mating and sporulation defects', 'Nucleic acid metabolism defects', 'Sensitivity to aminoacid analogs and other drugs', 'Sensitivity to antibiotics', 'Sensitivity to immunosuppressants', 'Stress response defects', 'Unknown'}
    Motif TEXT NOT NULL,  -- e.g. 'PS00824'
    Chromosome INTEGER NOT NULL,  -- e.g. 1
    Function TEXT NOT NULL,  -- values: {'CELL GROWTH, CELL DIVISION AND DNA SYNTHESIS', 'CELL RESCUE, DEFENSE, CELL DEATH AND AGEING', 'CELLULAR BIOGENESIS (proteins are not localized to the corresponding organelle)', 'CELLULAR COMMUNICATION/SIGNAL TRANSDUCTION', 'CELLULAR ORGANIZATION (proteins are localized to the corresponding organelle)', 'CELLULAR TRANSPORT AND TRANSPORTMECHANISMS', 'ENERGY', 'IONIC HOMEOSTASIS', 'METABOLISM', 'PROTEIN DESTINATION', 'PROTEIN SYNTHESIS', 'TRANSCRIPTION', 'TRANSPORT FACILITATION'}
    Localization TEXT NOT NULL,  -- values: {'ER', 'cell wall', 'cytoplasm', 'cytoskeleton', 'endosome', 'extracellular', 'golgi', 'integral membrane', 'lipid particles', 'mitochondria', 'nucleus', 'peroxisome', 'plasma membrane', 'transport vesicles', 'vacuole'}
    FOREIGN KEY (GeneID) REFERENCES Classification(GeneID)
);

-- Table: Interactions (910 rows)
CREATE TABLE Interactions (
    GeneID1 TEXT NOT NULL,  -- e.g. 'G234064'; FK -> Classification.GeneID
    GeneID2 TEXT NOT NULL,  -- e.g. 'G234126'; FK -> Classification.GeneID
    Type TEXT NOT NULL,  -- values: {'Genetic', 'Genetic-Physical', 'Physical'}
    Expression_Corr REAL NOT NULL,  -- e.g. 0.914
    PRIMARY KEY (GeneID1, GeneID2),
    FOREIGN KEY (GeneID1) REFERENCES Classification(GeneID),
    FOREIGN KEY (GeneID2) REFERENCES Classification(GeneID)
);
