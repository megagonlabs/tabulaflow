-- Database: ice_hockey_draft

-- Table: PlayerInfo (2171 rows)
CREATE TABLE PlayerInfo (
    ELITEID INTEGER PRIMARY KEY,  -- e.g. 9
    PlayerName TEXT,  -- e.g. 'David Bornhammar'
    birthdate TEXT,  -- e.g. '1981-06-15'
    birthyear DATE,  -- e.g. 1981
    birthmonth INTEGER,  -- e.g. 6
    birthday INTEGER,  -- e.g. 15
    birthplace TEXT,  -- e.g. 'Lidingo, SWE'
    nation TEXT,  -- e.g. 'Sweden'
    height INTEGER,  -- e.g. 73; FK -> height_info.height_id
    weight INTEGER,  -- e.g. 198; FK -> weight_info.weight_id
    position_info TEXT,  -- e.g. 'D'
    shoots TEXT,  -- values: {'-', 'L', 'R'}
    draftyear INTEGER,  -- e.g. 1999
    draftround INTEGER,  -- e.g. 7
    overall INTEGER,  -- e.g. 192
    overallby TEXT,  -- e.g. 'Washington Capitals'
    CSS_rank INTEGER,  -- e.g. 192
    sum_7yr_GP INTEGER,  -- e.g. 0
    sum_7yr_TOI INTEGER,  -- e.g. 0
    GP_greater_than_0 TEXT,  -- values: {'no', 'yes'}
    FOREIGN KEY (height) REFERENCES height_info(height_id),
    FOREIGN KEY (weight) REFERENCES weight_info(weight_id)
);

-- Table: SeasonStatus (5485 rows)
CREATE TABLE SeasonStatus (
    ELITEID INTEGER,  -- e.g. 3667; FK -> PlayerInfo.ELITEID
    SEASON TEXT,  -- values: {'1997-1998', '1998-1999', '1999-2000', '2000-2001', '2001-2002', '2003-2004', '2004-2005', '2005-2006', '2006-2007', '2007-2008'}
    TEAM TEXT,  -- e.g. 'Rimouski Oceanic'
    LEAGUE TEXT,  -- e.g. 'QMJHL'
    GAMETYPE TEXT,  -- values: {'Playoffs', 'Regular Season'}
    GP INTEGER,  -- e.g. 58
    G INTEGER,  -- e.g. 44
    A INTEGER,  -- e.g. 71
    P INTEGER,  -- e.g. 115
    PIM INTEGER,  -- e.g. 117
    PLUSMINUS INTEGER,  -- e.g. 27
    FOREIGN KEY (ELITEID) REFERENCES PlayerInfo(ELITEID)
);

-- Table: height_info (16 rows)
CREATE TABLE height_info (
    height_id INTEGER PRIMARY KEY,  -- e.g. 65
    height_in_cm INTEGER,  -- e.g. 165
    height_in_inch TEXT  -- e.g. '5'5"'
);

-- Table: weight_info (46 rows)
CREATE TABLE weight_info (
    weight_id INTEGER PRIMARY KEY,  -- e.g. 154
    weight_in_kg INTEGER,  -- e.g. 70
    weight_in_lbs INTEGER  -- e.g. 154
);
