-- Table: all_prevalences (244 rows)
CREATE TABLE all_prevalences (
    ITEM TEXT PRIMARY KEY,  -- e.g. '0.3 Ml Epinephrine 0.5 Mg/Ml Auto Injector'
    "POPULATION TYPE" TEXT,  -- values: {'LIVING'}
    OCCURRENCES INTEGER,  -- e.g. 868
    "POPULATION COUNT" INTEGER,  -- e.g. 1000
    "PREVALENCE RATE" REAL,  -- e.g. 0.868
    "PREVALENCE PERCENTAGE" REAL  -- e.g. 86.800
);

-- Table: allergies (572 rows)
CREATE TABLE allergies (
    START TEXT,  -- e.g. '3/11/95'
    STOP TEXT,  -- e.g. '12/22/14'
    PATIENT TEXT,  -- e.g. '00341a88-1cc1-4b39-b0f9-05b0531991a0'; FK -> patients.patient
    ENCOUNTER TEXT,  -- e.g. 'ddc8a625-07e6-4a02-a616-e0ce07b5f305'; FK -> encounters.ID
    CODE INTEGER,  -- e.g. 232347008
    DESCRIPTION TEXT,  -- e.g. 'Allergy to dairy product'
    PRIMARY KEY (PATIENT, ENCOUNTER, CODE),
    FOREIGN KEY (ENCOUNTER) REFERENCES encounters(ID),
    FOREIGN KEY (PATIENT) REFERENCES patients(patient)
);
