-- Table: Match (577 rows)
CREATE TABLE Match (
    Match_Id INTEGER PRIMARY KEY,  -- e.g. 335987
    Team_1 INTEGER,  -- e.g. 2; FK -> Team.Team_Id
    Team_2 INTEGER,  -- e.g. 1; FK -> Team.Team_Id
    Match_Date DATE,  -- e.g. '2008-04-18'
    Season_Id INTEGER,  -- e.g. 1; FK -> Season.Season_Id
    Venue_Id INTEGER,  -- e.g. 1; FK -> Venue.Venue_Id
    Toss_Winner INTEGER,  -- e.g. 2; FK -> Team.Team_Id
    Toss_Decide INTEGER,  -- e.g. 1; FK -> Toss_Decision.Toss_Id
    Win_Type INTEGER,  -- e.g. 1; FK -> Win_By.Win_Id
    Win_Margin INTEGER,  -- e.g. 140
    Outcome_type INTEGER,  -- e.g. 1; FK -> Out_Type.Out_Id
    Match_Winner INTEGER,  -- e.g. 1; FK -> Team.Team_Id
    Man_of_the_Match INTEGER,  -- e.g. 2; FK -> Player.Player_Id
    FOREIGN KEY (Team_1) REFERENCES Team(Team_Id),
    FOREIGN KEY (Team_2) REFERENCES Team(Team_Id),
    FOREIGN KEY (Season_Id) REFERENCES Season(Season_Id),
    FOREIGN KEY (Venue_Id) REFERENCES Venue(Venue_Id),
    FOREIGN KEY (Toss_Winner) REFERENCES Team(Team_Id),
    FOREIGN KEY (Toss_Decide) REFERENCES Toss_Decision(Toss_Id),
    FOREIGN KEY (Win_Type) REFERENCES Win_By(Win_Id),
    FOREIGN KEY (Outcome_type) REFERENCES Out_Type(Out_Id),
    FOREIGN KEY (Match_Winner) REFERENCES Team(Team_Id),
    FOREIGN KEY (Man_of_the_Match) REFERENCES Player(Player_Id)
);
