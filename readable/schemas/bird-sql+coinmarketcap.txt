-- Database: coinmarketcap

-- Table: coins (8927 rows)
CREATE TABLE coins (
    id INTEGER NOT NULL PRIMARY KEY,  -- e.g. 1
    name TEXT,  -- e.g. 'Bitcoin'
    slug TEXT,  -- e.g. 'bitcoin'
    symbol TEXT,  -- e.g. 'BTC'
    status TEXT,  -- values: {'active', 'extinct', 'inactive', 'untracked'}
    category TEXT,  -- values: {'coin', 'token'}
    description TEXT,  -- e.g. '## **What Is Bitcoin (BTC)?**\n\nBitcoin is a dece...ple, using the alias [Satoshi Nakamoto](https://co'
    subreddit TEXT,  -- e.g. 'bitcoin'
    notice TEXT,  -- e.g. 'TIPS has undergone a [swap](https://bitcointalk.or...//coinmarketcap.com/currencies/fedora-gold) (FED).'
    tags TEXT,  -- e.g. 'mineable, pow, sha-256, store-of-value, state-chan...apital-portfolio, boostvc-portfolio, cms-holdings-'
    tag_names TEXT,  -- e.g. 'Mineable, PoW, SHA-256, Store of Value, State chan...apital Portfolio, BoostVC Portfolio, CMS Holdings '
    website TEXT,  -- e.g. 'https://bitcoin.org/'
    platform_id INTEGER,  -- e.g. 1027
    date_added TEXT,  -- e.g. '2013-04-28T00:00:00.000Z'
    date_launched TEXT  -- e.g. '2018-01-16T00:00:00.000Z'
);

-- Table: historical (4441972 rows)
CREATE TABLE historical (
    date DATE,  -- e.g. '2013-04-28'
    coin_id INTEGER,  -- e.g. 1
    cmc_rank INTEGER,  -- e.g. 1
    market_cap REAL,  -- e.g. 1488566971.956
    price REAL,  -- e.g. 134.210
    open REAL,  -- e.g. 134.444
    high REAL,  -- e.g. 147.488
    low REAL,  -- e.g. 134.000
    close REAL,  -- e.g. 144.540
    time_high TEXT,  -- e.g. '13:15:01'
    time_low TEXT,  -- e.g. '05:20:01'
    volume_24h REAL,  -- e.g. 0.000
    percent_change_1h REAL,  -- e.g. 0.639
    percent_change_24h REAL,  -- e.g. 7.636
    percent_change_7d REAL,  -- e.g. -13.809
    circulating_supply REAL,  -- e.g. 11091325.000
    total_supply REAL,  -- e.g. 11091325.000
    max_supply REAL,  -- e.g. 21000000.000
    num_market_pairs INTEGER  -- e.g. 7050
);
