-- Database: movie_platform

-- Table: lists (79565 rows)
CREATE TABLE lists (
    user_id INTEGER,  -- e.g. 88260493; FK -> lists_users.user_id
    list_id INTEGER NOT NULL PRIMARY KEY,  -- e.g. 1
    list_title TEXT,  -- e.g. 'Films that made your kid sister cry'
    list_movie_number INTEGER,  -- e.g. 5
    list_update_timestamp_utc TEXT,  -- e.g. '2019-01-24 19:16:18'
    list_creation_timestamp_utc TEXT,  -- e.g. '2009-11-11 00:02:21'
    list_followers INTEGER,  -- e.g. 5
    list_url TEXT,  -- e.g. 'http://mubi.com/lists/films-that-made-your-kid-sister-cry'
    list_comments INTEGER,  -- e.g. 3
    list_description TEXT,  -- e.g. '<p>Don’t be such a baby!!</p>
<p><strong>bold</strong></p>'
    list_cover_image_url TEXT,  -- e.g. 'https://assets.mubicdn.net/images/film/3822/image-w1280.jpg?1445914994'
    list_first_image_url TEXT,  -- e.g. 'https://assets.mubicdn.net/images/film/3822/image-w320.jpg?1445914994'
    list_second_image_url TEXT,  -- e.g. 'https://assets.mubicdn.net/images/film/506/image-w320.jpg?1543838422'
    list_third_image_url TEXT,  -- e.g. 'https://assets.mubicdn.net/images/film/485/image-w320.jpg?1575331204'
    FOREIGN KEY (user_id) REFERENCES lists_users(user_id)
);

-- Table: lists_users (80311 rows)
CREATE TABLE lists_users (
    user_id INTEGER NOT NULL,  -- e.g. 2385; FK -> lists.user_id
    list_id INTEGER NOT NULL,  -- e.g. 192287; FK -> lists.list_id
    list_update_date_utc TEXT,  -- e.g. '2019-11-26'
    list_creation_date_utc TEXT,  -- e.g. '2009-12-18'
    user_trialist INTEGER,  -- e.g. 1
    user_subscriber INTEGER,  -- e.g. 1
    user_avatar_image_url TEXT,  -- e.g. 'https://assets.mubicdn.net/images/avatars/74983/images-w150.jpg?1523895214'
    user_cover_image_url TEXT,  -- e.g. 'https://assets.mubicdn.net/images/cover_images/12788/images-small.jpg?1406887796'
    user_eligible_for_trial TEXT,  -- values: {'0', '1'}
    user_has_payment_method TEXT,  -- values: {'0', '1'}
    PRIMARY KEY (user_id, list_id),
    FOREIGN KEY (list_id) REFERENCES lists(list_id),
    FOREIGN KEY (user_id) REFERENCES lists(user_id)
);

-- Table: movies (226087 rows)
CREATE TABLE movies (
    movie_id INTEGER NOT NULL PRIMARY KEY,  -- e.g. 1
    movie_title TEXT,  -- e.g. 'La Antena'
    movie_release_year INTEGER,  -- e.g. 2007
    movie_url TEXT,  -- e.g. 'http://mubi.com/films/la-antena'
    movie_title_language TEXT,  -- values: {'en'}
    movie_popularity INTEGER,  -- e.g. 105
    movie_image_url TEXT,  -- e.g. 'https://images.mubicdn.net/images/film/1/cache-7927-1581389497/image-w1280.jpg'
    director_id TEXT,  -- e.g. '131'
    director_name TEXT,  -- e.g. 'Esteban Sapir'
    director_url TEXT  -- e.g. 'http://mubi.com/cast/esteban-sapir'
);

-- Table: ratings (15517252 rows)
CREATE TABLE ratings (
    movie_id INTEGER,  -- e.g. 1066; FK -> movies.movie_id
    rating_id INTEGER,  -- e.g. 15610495; FK -> ratings.rating_id
    rating_url TEXT,  -- e.g. 'http://mubi.com/films/pavee-lackeen-the-traveller-girl/ratings/15610495'
    rating_score INTEGER,  -- e.g. 3
    rating_timestamp_utc TEXT,  -- e.g. '2017-06-10 12:38:33'
    critic TEXT,  -- e.g. 'I am a bit disappointed by this documentary film I...king. Otherwise it is O.K. and rather interesting.'
    critic_likes INTEGER,  -- e.g. 0
    critic_comments INTEGER,  -- e.g. 0
    user_id INTEGER,  -- e.g. 41579158; FK -> lists_users.user_id; FK -> ratings_users.user_id
    user_trialist INTEGER,  -- e.g. 0
    user_subscriber INTEGER,  -- e.g. 0
    user_eligible_for_trial INTEGER,  -- e.g. 1
    user_has_payment_method INTEGER,  -- e.g. 0
    FOREIGN KEY (movie_id) REFERENCES movies(movie_id),
    FOREIGN KEY (user_id) REFERENCES lists_users(user_id),
    FOREIGN KEY (rating_id) REFERENCES ratings(rating_id),
    FOREIGN KEY (user_id) REFERENCES ratings_users(user_id)
);

-- Table: ratings_users (4297641 rows)
CREATE TABLE ratings_users (
    user_id INTEGER,  -- e.g. 41579158; FK -> lists_users.user_id
    rating_date_utc TEXT,  -- e.g. '2017-06-10'
    user_trialist INTEGER,  -- e.g. 0
    user_subscriber INTEGER,  -- e.g. 0
    user_avatar_image_url TEXT,  -- e.g. 'https://assets.mubicdn.net/images/avatars/74283/images-w150.jpg?1523895155'
    user_cover_image_url TEXT,  -- e.g. 'https://assets.mubicdn.net/images/cover_images/37844/images-small.png?1444549039'
    user_eligible_for_trial INTEGER,  -- e.g. 1
    user_has_payment_method INTEGER,  -- e.g. 0
    FOREIGN KEY (user_id) REFERENCES lists_users(user_id)
);
