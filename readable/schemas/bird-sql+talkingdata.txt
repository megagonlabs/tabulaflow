-- Table: app_all (113211 rows)
CREATE TABLE app_all (
    app_id INTEGER NOT NULL PRIMARY KEY  -- e.g. -9223281467940916832
);

-- Table: app_events (32473067 rows)
CREATE TABLE app_events (
    event_id INTEGER NOT NULL,  -- e.g. 2; FK -> events.event_id
    app_id INTEGER NOT NULL,  -- e.g. -8942695423876075857
    is_installed INTEGER NOT NULL,  -- e.g. 1
    is_active INTEGER NOT NULL,  -- e.g. 0
    PRIMARY KEY (event_id, app_id),
    FOREIGN KEY (event_id) REFERENCES events(event_id)
);

-- Table: app_events_relevant (3701900 rows)
CREATE TABLE app_events_relevant (
    event_id INTEGER NOT NULL,  -- e.g. 2; FK -> events_relevant.event_id
    app_id INTEGER NOT NULL,  -- e.g. -8942695423876075857; FK -> app_all.app_id
    is_installed INTEGER,  -- e.g. 1
    is_active INTEGER,  -- e.g. 0
    PRIMARY KEY (event_id, app_id),
    FOREIGN KEY (app_id) REFERENCES app_all(app_id),
    FOREIGN KEY (event_id) REFERENCES events_relevant(event_id)
);

-- Table: app_labels (459943 rows)
CREATE TABLE app_labels (
    app_id INTEGER NOT NULL,  -- e.g. 7324884708820027918; FK -> app_all.app_id
    label_id INTEGER NOT NULL,  -- e.g. 251; FK -> label_categories.label_id
    FOREIGN KEY (app_id) REFERENCES app_all(app_id),
    FOREIGN KEY (label_id) REFERENCES label_categories(label_id)
);

-- Table: events (3252950 rows)
CREATE TABLE events (
    event_id INTEGER NOT NULL PRIMARY KEY,  -- e.g. 1
    device_id INTEGER,  -- e.g. 29182687948017175
    timestamp DATETIME,  -- e.g. '2016-05-01 00:55:25.0'
    longitude REAL,  -- e.g. 121.000
    latitude REAL  -- e.g. 31.000
);

-- Table: events_relevant (167389 rows)
CREATE TABLE events_relevant (
    event_id INTEGER NOT NULL PRIMARY KEY,  -- e.g. 2
    device_id INTEGER,  -- FK -> gender_age.device_id

    longitude REAL NOT NULL,  -- e.g. 1.000
    latitude REAL NOT NULL,  -- e.g. 0.000
    FOREIGN KEY (device_id) REFERENCES gender_age(device_id)
);
