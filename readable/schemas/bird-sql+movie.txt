-- Database: movie

-- Table: actor (2713 rows)
CREATE TABLE actor (
    ActorID INTEGER PRIMARY KEY,  -- e.g. 1
    Name TEXT,  -- e.g. 'John Travolta'
    "Date of Birth" DATE,  -- e.g. '1954-02-18'
    "Birth City" TEXT,  -- e.g. 'Englewood'
    "Birth Country" TEXT,  -- e.g. 'USA'
    "Height (Inches)" INTEGER,  -- e.g. 74
    Biography TEXT,  -- e.g. 'John Joseph Travolta was born in Englewood, New Je.... Travolta started acting appearing in a local ...'
    Gender TEXT,  -- values: {'Female', 'Male'}
    Ethnicity TEXT,  -- values: {'African American', 'American-Irish', 'American-South African', 'Argentine', 'Armenian', 'Asian American', 'Asian', 'Israeli', 'Japanese-American', 'Latino', 'Lebanese', 'Mixed', 'Native American', 'Polish', 'Puerto Rican', 'South African', 'White'}
    NetWorth TEXT  -- e.g. '$250,000,000.00'
);

-- Table: characters (4312 rows)
CREATE TABLE characters (
    MovieID INTEGER,  -- e.g. 1; FK -> movie.MovieID
    ActorID INTEGER,  -- e.g. 1; FK -> actor.ActorID
    "Character Name" TEXT,  -- e.g. 'James'
    creditOrder INTEGER,  -- e.g. 1
    pay TEXT,  -- values: {'$100,000,000.00', '$29,000,000.00', '$5,000,000.00', '$50,000,000.00', '$55,000,000.00', '$60,000,000.00', '$65,000,000.00', '$68,000,000.00', '$70,000,000.00', '$75,000,000.00'}
    screentime TEXT,  -- values: {'0:02:45', '0:03:30', '0:04:15', '0:04:30', '0:08:45', '0:29:00', '0:32:15', '0:32:30'}
    PRIMARY KEY (MovieID, ActorID),
    FOREIGN KEY (ActorID) REFERENCES actor(ActorID),
    FOREIGN KEY (MovieID) REFERENCES movie(MovieID)
);

-- Table: movie (634 rows)
CREATE TABLE movie (
    MovieID INTEGER PRIMARY KEY,  -- e.g. 1
    Title TEXT,  -- e.g. 'Look Who's Talking'
    "MPAA Rating" TEXT,  -- values: {'G', 'PG', 'PG-13', 'R'}
    Budget INTEGER,  -- e.g. 7500000
    Gross INTEGER,  -- e.g. 296000000
    "Release Date" TEXT,  -- e.g. '1989-10-12'
    Genre TEXT,  -- e.g. 'Romance'
    Runtime INTEGER,  -- e.g. 93
    Rating REAL,  -- e.g. 5.900
    "Rating Count" INTEGER,  -- e.g. 73638
    Summary TEXT  -- e.g. 'After a single, career-minded woman is left on her...of the newborn boy is narrated through voice-over.'
);
