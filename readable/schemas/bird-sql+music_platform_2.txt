-- Database: music_platform_2

-- Table: categories (210329 rows)
CREATE TABLE categories (
    podcast_id TEXT NOT NULL,  -- e.g. 'a00018b54eb342567c94dacfb2a3e504'; FK -> podcasts.podcast_id
    category TEXT NOT NULL,  -- e.g. 'arts'
    PRIMARY KEY (podcast_id, category),
    FOREIGN KEY (podcast_id) REFERENCES podcasts(podcast_id)
);

-- Table: podcasts (108578 rows)
CREATE TABLE podcasts (
    podcast_id TEXT PRIMARY KEY,  -- e.g. 'a00018b54eb342567c94dacfb2a3e504'
    itunes_id INTEGER NOT NULL,  -- e.g. 1313466221
    slug TEXT NOT NULL,  -- e.g. 'scaling-global'
    itunes_url TEXT NOT NULL,  -- e.g. 'https://podcasts.apple.com/us/podcast/scaling-global/id1313466221'
    title TEXT NOT NULL  -- e.g. 'Scaling Global'
);

-- Table: reviews (1964856 rows)
CREATE TABLE reviews (
    podcast_id TEXT NOT NULL,  -- e.g. 'c61aa81c9b929a66f0c1db6cbe5d8548'; FK -> podcasts.podcast_id
    title TEXT NOT NULL,  -- e.g. 'really interesting!'
    content TEXT NOT NULL,  -- e.g. 'Thanks for providing these insights.  Really enjoy the variety and depth -- please keep them coming!'
    rating INTEGER NOT NULL,  -- e.g. 5
    author_id TEXT NOT NULL,  -- e.g. 'F7E5A318989779D'
    created_at TEXT NOT NULL,  -- e.g. '2018-04-24T12:05:16-07:00'
    FOREIGN KEY (podcast_id) REFERENCES podcasts(podcast_id)
);

-- Table: runs (12 rows)
CREATE TABLE runs (
    run_at TEXT NOT NULL,  -- e.g. '2021-05-10 02:53:00'
    max_rowid INTEGER NOT NULL,  -- e.g. 3266481
    reviews_added INTEGER NOT NULL  -- e.g. 1215223
);
