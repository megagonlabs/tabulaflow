-- Database: simpson_episodes

-- Table: Award (75 rows)
CREATE TABLE Award (
    award_id INTEGER PRIMARY KEY,  -- e.g. 325
    organization TEXT,  -- e.g. 'Primetime Emmy Awards'
    year INTEGER,  -- e.g. 2009
    award_category TEXT,  -- e.g. 'Primetime Emmy'
    award TEXT,  -- e.g. 'Outstanding Voice-Over Performance'
    person TEXT,  -- e.g. 'Dan Castellaneta'; FK -> Person.name
    role TEXT,  -- e.g. 'executive producer'
    episode_id TEXT,  -- e.g. 'S20-E18'; FK -> Episode.episode_id
    season TEXT,
    song TEXT,
    result TEXT,  -- values: {'Nominee', 'Winner'}
    FOREIGN KEY (person) REFERENCES Person(name),
    FOREIGN KEY (episode_id) REFERENCES Episode(episode_id)
);

-- Table: Character_Award (12 rows)
CREATE TABLE Character_Award (
    award_id INTEGER,  -- e.g. 325; FK -> Award.award_id
    character TEXT,  -- values: {'Homer Simpson', 'Kent Brockman', 'Lenny', 'Moe Szyslak', 'Mr. Burns', 'Smithers'}
    FOREIGN KEY (award_id) REFERENCES Award(award_id)
);

-- Table: Credit (4557 rows)
CREATE TABLE Credit (
    episode_id TEXT,  -- e.g. 'S20-E10'; FK -> Episode.episode_id
    category TEXT,  -- values: {'Animation Department', 'Art Department', 'Cast', 'Casting Department', 'Directed by', 'Editorial Department', 'General', 'Music Department', 'Other crew', 'Produced by', 'Production Management', 'Script and Continuity Department', 'Second Unit Director or Assistant Director', 'Sound Department', 'Thanks', 'Visual Effects by', 'Writing Credits'}
    person TEXT,  -- e.g. 'Bonita Pietila'; FK -> Person.name
    role TEXT,  -- e.g. 'casting'
    credited TEXT,  -- values: {'false', 'true'}
    FOREIGN KEY (episode_id) REFERENCES Episode(episode_id),
    FOREIGN KEY (person) REFERENCES Person(name)
);

-- Table: Episode (21 rows)
CREATE TABLE Episode (
    episode_id TEXT PRIMARY KEY,  -- e.g. 'S20-E1'
    season INTEGER,  -- e.g. 20
    episode INTEGER,  -- e.g. 1
    number_in_series INTEGER,  -- e.g. 421
    title TEXT,  -- e.g. 'Sex, Pies and Idiot Scrapes'
    summary TEXT,  -- e.g. 'Homer and Ned go into business together as bounty ...unters, and Marge takes a job at an erotic bakery.'
    air_date TEXT,  -- e.g. '2008-09-28'
    episode_image TEXT,  -- e.g. 'https://m.media-amazon.com/images/M/MV5BMTYwMzk2Nj...FtZTgwMzA2MDQ2MjE@._V1_UX224_CR0,0,224,126_AL_.jpg'
    rating REAL,  -- e.g. 7.200
    votes INTEGER  -- e.g. 1192
);

-- Table: Keyword (307 rows)
CREATE TABLE Keyword (
    episode_id TEXT,  -- e.g. 'S20-E1'; FK -> Episode.episode_id
    keyword TEXT,  -- e.g. '1930s to 2020s'
    PRIMARY KEY (episode_id, keyword),
    FOREIGN KEY (episode_id) REFERENCES Episode(episode_id)
);

-- Table: Person (369 rows)
CREATE TABLE Person (
    name TEXT PRIMARY KEY,  -- e.g. 'Adam Greeley'
    birthdate TEXT,  -- e.g. '1963-05-04'
    birth_name TEXT,  -- e.g. 'Marc Edward Wilmore'
    birth_place TEXT,  -- e.g. 'USA'
    birth_region TEXT,  -- e.g. 'California'
    birth_country TEXT,  -- values: {'Canada', 'Czechoslovakia', 'France', 'Iran', 'Ireland', 'North Korea', 'Philippines', 'UK', 'USA'}
    height_meters REAL,  -- e.g. 1.850
    nickname TEXT  -- e.g. 'Jim'
);

-- Table: Vote (210 rows)
CREATE TABLE Vote (
    episode_id TEXT,  -- e.g. 'S20-E1'; FK -> Episode.episode_id
    stars INTEGER,  -- e.g. 2
    votes INTEGER,  -- e.g. 16
    percent REAL,  -- e.g. 1.300
    FOREIGN KEY (episode_id) REFERENCES Episode(episode_id)
);
